> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `Gallery` nav link and a `/gallery` page that loads public Google Drive images in a grid with skeleton loading and infinite scroll.

**Architecture:** Client-side `/gallery` page fetches the Google Drive API `files.list` using a public API key to list image files in a public folder. A dedicated gallery data module handles request + pagination mapping, while UI components render the grid, skeletons, and error/empty states.

**Tech Stack:** Next.js App Router, React, Tailwind CSS, framer-motion, Vitest

```ts
import type { DriveFile, GalleryItem, GalleryPage } from '@/types'

interface DriveResponse {
  nextPageToken?: string
  files: DriveFile[]
}

interface BuildDriveUrlOptions {
  folderId: string
  apiKey: string
  pageToken?: string
  pageSize?: number
}

const DEFAULT_PAGE_SIZE = 24

export const buildDriveUrl = ({ folderId, apiKey, pageToken, pageSize = DEFAULT_PAGE_SIZE }: BuildDriveUrlOptions) => {
  const params = new URLSearchParams({
    q: `'${folderId}' in parents and mimeType contains 'image/' and trashed=false`,
    orderBy: 'modifiedTime desc',
    pageSize: String(pageSize),
    fields: 'nextPageToken, files(id, name, modifiedTime, thumbnailLink, webViewLink)',
    key: apiKey,
  })

  if (pageToken) params.set('pageToken', pageToken)

  return `https://www.googleapis.com/drive/v3/files?${params.toString()}`
}

export const mapDriveResponse = (response: DriveResponse): GalleryPage => {
  const items: GalleryItem[] = response.files.map((file) => ({
    id: file.id,
    name: file.name,
    modifiedTime: file.modifiedTime,
    thumbnailUrl: file.thumbnailLink,
    viewUrl: `https://drive.google.com/uc?export=view&id=${file.id}`,
  }))

  return { items, nextPageToken: response.nextPageToken }
}

export const fetchGalleryPage = async ({ folderId, apiKey, pageToken }: { folderId: string; apiKey: string; pageToken?: string }) => {
  const url = buildDriveUrl({ folderId, apiKey, pageToken })
  const response = await fetch(url)

  if (!response.ok) {
    throw new Error(`Drive API error: ${response.status}`)
  }

  const data = (await response.json()) as DriveResponse
  return mapDriveResponse(data)
}
```

```tsx
import type { GalleryItem } from '@/types'
import { GalleryCard } from './gallery-card'
import { GallerySkeleton } from './gallery-skeleton'
import { useEffect, useRef } from 'react'

interface GalleryGridProps {
  items: GalleryItem[]
  hasMore: boolean
  isLoading: boolean
  onLoadMore: () => void
}

export const GalleryGrid = ({ items, hasMore, isLoading, onLoadMore }: GalleryGridProps) => {
  const sentinelRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (!hasMore || isLoading) return

    const sentinel = sentinelRef.current
    if (!sentinel) return

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) onLoadMore()
      },
      { rootMargin: '200px' }
    )

    observer.observe(sentinel)
    return () => observer.disconnect()
  }, [hasMore, isLoading, onLoadMore])

  return (
    <div className="space-y-6">
      <div className="grid grid-cols-2 md:grid-cols-3 gap-4">
        {items.map((item) => (
          <GalleryCard key={item.id} item={item} />
        ))}
      </div>

      {isLoading && <GallerySkeleton count={12} />}

      <div ref={sentinelRef} aria-hidden="true" />
    </div>
  )
}
```

```tsx
import type { Metadata } from 'next'
import { BreadcrumbSchema, JsonLd } from '@/components/seo/json-ld'
import { Footer } from '@/components/footer'
import { absoluteUrl, siteConfig } from '@/lib/site'
import { constructMetadata } from '@/lib/seo'
import { Camera } from 'lucide-react'
import { GalleryGrid } from './_components/gallery-grid'
import { useCallback, useEffect, useState } from 'react'
import type { GalleryItem } from '@/types'
import { fetchGalleryPage } from '@/lib/gallery'

export const metadata: Metadata = constructMetadata({
  title: 'Gallery | MD Ashikur Rahman Puspo',
  description: 'A curated gallery of photos and moments by MD Ashikur Rahman Puspo.',
  url: absoluteUrl('/gallery'),
  keywords: ['Gallery', 'Photography', 'Moments', 'Portfolio']
})

const FOLDER_ID = '13GV6w5NOoOJ_Mf44ho9082swYTKKMORy'

export default function GalleryPage() {
  const [items, setItems] = useState<GalleryItem[]>([])
  const [nextPageToken, setNextPageToken] = useState<string | undefined>(undefined)
  const [isLoading, setIsLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const loadPage = useCallback(async () => {
    const apiKey = process.env.NEXT_PUBLIC_GOOGLE_DRIVE_API_KEY
    if (!apiKey) {
      setError('Missing Google Drive API key.')
      return
    }

    setIsLoading(true)
    setError(null)
    try {
      const page = await fetchGalleryPage({ folderId: FOLDER_ID, apiKey, pageToken: nextPageToken })
      setItems((prev) => [...prev, ...page.items])
      setNextPageToken(page.nextPageToken)
    } catch (err) {
      console.error(err)
      setError('Unable to load gallery. Please try again.')
    } finally {
      setIsLoading(false)
    }
  }, [nextPageToken])

  useEffect(() => {
    loadPage()
  }, [])

  const hasMore = Boolean(nextPageToken)

  const galleryPageJsonLd = {
    '@context': 'https://schema.org',
    '@type': 'CollectionPage',
    '@id': `${absoluteUrl('/gallery')}#webpage`,
    url: absoluteUrl('/gallery'),
    name: 'Gallery | MD Ashikur Rahman Puspo',
    description: 'Photo gallery by MD Ashikur Rahman Puspo.',
    isPartOf: { '@id': `${siteConfig.url}/#website` },
    about: { '@id': `${siteConfig.url}/#person` },
  }

  return (
    <main className="min-h-screen bg-background">
      <JsonLd data={galleryPageJsonLd} />
      <BreadcrumbSchema items={[{ name: 'Home', item: '/' }, { name: 'Gallery', item: '/gallery' }]} />
      <div className="container mx-auto px-6 max-w-5xl pt-28 pb-20">
        <div className="mb-16">
          <div className="flex items-start gap-4">
            <div className="p-3 rounded-2xl bg-emerald-500/10 border border-emerald-500/20 shrink-0">
              <Camera className="w-8 h-8 text-emerald-500" />
            </div>
            <div>
              <h1 className="text-4xl md:text-5xl font-bold mb-3 tracking-tight">
                <span className="text-transparent bg-clip-text bg-gradient-to-r from-foreground via-foreground/90 to-foreground/60">
                  Gallery
                </span>
              </h1>
              <p className="text-muted-foreground max-w-2xl text-base md:text-lg leading-relaxed">
                A curated collection of photos and moments.
              </p>
            </div>
          </div>
        </div>

        {error && (
          <div className="mb-6 rounded-xl border border-rose-500/20 bg-rose-500/10 p-4 text-sm text-rose-600">
            {error}
            <button
              type="button"
              className="ml-3 text-rose-700 underline"
              onClick={loadPage}
            >
              Retry
            </button>
          </div>
        )}

        {items.length === 0 && !isLoading && !error && (
          <p className="text-muted-foreground">No photos yet.</p>
        )}

        <GalleryGrid
          items={items}
          hasMore={hasMore}
          isLoading={isLoading}
          onLoadMore={loadPage}
        />
      </div>
      <Footer />
    </main>
  )
}
```
